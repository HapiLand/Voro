using System.Collections.Generic;
using UnityEngine;
using VoroSystem.TerrainGeneration.GenerationSys.Results;

namespace VoroSystem.TerrainGeneration.GenerationSys.Effects {
public class ConstantEffect : EffectBase {
    float _amount;
    bool _overwriteExisting;

    public ConstantEffect(ControlContainer controls) : base(controls) {
        // set the EffectCompute to use the shader for this Effect
        ShaderCompute = new EffectCompute("FXConstant");

        // set the parameters
        SetShaderParameters();
    }

    public override NodeType Type => NodeType.Constant;

    public override void SetShaderParameters() {
        _overwriteExisting = (bool)Controls.GetControl("Overwrite Existing").Value;
        _amount = (float)Controls.GetControl("Amount").Value;
    }

    public override void Dispatch(Result result) {
        // set the PointBuffer in the shader
        base.Dispatch(result);

        // set parameters for the shader
        var parameters = new Dictionary<string, object>
        {
            { "overwrite_existing", _overwriteExisting },
            { "amount", _amount }
        };

        // execute the Shader to perform the generation function
        ShaderCompute.Dispatch(parameters, result.PointBuffer.count);
        Debug.Log("dispatch success");
    }
}
