using UnityEngine;
using VoroSystem.GraphEditor;
using VoroSystem.TerrainGeneration.GenerationSys;
using VoroSystem.WorldGrid;

namespace VoroSystem.TerrainGeneration {
/// <summary>
///     TerrainDesigner must be capable of:
///     Creating a Diagram
///     Computing a Result
///     Generating GameObjects
/// </summary>
public interface ITerrainGenSystem {
    void RunGeneration();
    void InstantiateGameObjects();
}

[ExecuteAlways]
public class TerrainDesigner : MonoBehaviour, ITerrainGenSystem {
    GenerationSystem _generationSystem;
    IGraphEditorSystem _graphSystem;
    IWorld _world;

    void Awake() {
        _world = GetComponent<WorldMap>();
        _graphSystem = GetComponent<GraphDesigner>();

        _generationSystem = new GenerationSystem(_graphSystem, _world);
    }

    /// <summary>
    ///     executes the generation process
    /// </summary>
    public void RunGeneration() {
        // generate the Graphs in order for the system to have Effects to compute
        _generationSystem.GenerateDiagramAndComputeResult();
    }

    public void InstantiateGameObjects() {
        var parent = transform;

        _world.ForEach(tile => {
            // create a GameObject that will be the parent object for the Chunk
            var tileWorldPosition = tile.WorldPosition;
            var obj = new GameObject("Chunk")
            {
                transform =
                {
                    position = tileWorldPosition
                }
            };
            obj.transform.SetParent(parent);

            // iterate through every Cell in the Chunk
            tile.Chunk.ForEach(cell => {
                // create a GameObject for every Chunk Cell
                var instance = new GameObject("Cell")
                {
                    transform =
                    {
                        position = tile.Chunk.CellPosition(cell) + tile.WorldPosition
                    }
                };
                instance.transform.SetParent(obj.transform);

                var meshFilter = instance.AddComponent<MeshFilter>();
                var meshRenderer = instance.AddComponent<MeshRenderer>();

                // set mesh in object
                var variant = 0;
                var ID = cell.ID;
                meshFilter.sharedMesh = Resources.Load<Mesh>($"Mesh/{ID}_{variant}");

                // set material
                var originalMat = Resources.Load<Material>("FbxMat");
                var mat = new Material(originalMat)
                {
                    color = cell.Color
                };
                meshRenderer.material = mat;
            });
        });
    }
}
}
