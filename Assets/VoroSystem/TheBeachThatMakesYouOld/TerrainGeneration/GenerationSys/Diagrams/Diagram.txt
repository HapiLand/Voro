using System;
using System.Collections.Generic;
using UnityEngine;
using VoroSystem.EditorGraph.GraphSys.Graphs;
using VoroSystem.TerrainGeneration.GenerationSys.Effects;
using VoroSystem.TerrainGeneration.GenerationSys.Results;
using VoroSystem.WorldGrid;

namespace VoroSystem.TerrainGeneration.GenerationSys.Diagrams {
public class Diagram : IDiagram {
    public Diagram(List<Graph> graphs, IWorld worldMapSystem) {
        // set Graphs so the Diagram contains Effects
        Items = graphs;

        // set Grid so the Diagram can use Chunks
        Map = worldMapSystem;

        // define how the Compute function behaves
        Compute = (graphList, chunkList) => {
            var funcReturn = new List<Result>();

            // prepare all the Results that will be Computed
            // a Result is made per Chunk, Chunk has Points
            chunkList.ForEach(tile => { funcReturn.Add(ResultFactory.PrepareNewResult(tile)); });

            // for each Graph, iterate through its Effects so each can be dispatched
            graphList.ForEach(graph => { graph.ForEach(fx => { RunEffect(fx, funcReturn); }); });

            // todo fix bug where only a single chunk of points gets modified
            //  setting GridFactory.CreateGrid(10, 10); causes that chunks points to be moved higher than
            //  setting GridFactory.CreateGrid(1, 1);
            //  maybe only a single PointBuffer is made and shared between everything

            // carries out this Effect to compute every Result
            void RunEffect(EffectBase effect, List<Result> results) {
                Debug.Log($"run Effect '{effect.Type.ToString()}' to compute {results.Count} Results");

                // set the PointBuffer in the Effect
                results.ForEach(effect.SetPointBuffer);

                // dispatch the effect to compute the Result
                results.ForEach(effect.Dispatch);
                Debug.Log($"{results.Count} Results successfully computed");
            }

            // get the data from the PointBuffer within each Result
            funcReturn.ForEach(r => r.ReadBuffer());
            // release the buffers in the results to free it from memory
            funcReturn.ForEach(r => r.PointBuffer.Release());

            return funcReturn;
        };
    }

    public List<Graph> Items { get; set; }

    public void ForEach(Action<Graph> action) {
        Items.ForEach(action);
    }

    public void Add(Graph item) {
        Items.Add(item);
    }

    public void Remove(Graph item) { }
    public void MoveUp(Graph item) { }
    public void MoveDown(Graph item) { }
    public IWorld Map { get; set; }

    public Func<List<Graph>, List<ITile>, List<Result>> Compute { get; set; }

    public List<Graph> GetGraphs() {
        var graphs = new List<Graph>();
        ForEach(graphs.Add);
        return graphs;
    }

    public List<ITile> GetTiles() {
        var tiles = new List<ITile>();
        ForEachTile(tiles.Add);
        return tiles;
    }

    public void ForEachTile(Action<ITile> action) {
        Map.ForEach(action);
    }
}
}
